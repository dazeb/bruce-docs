## 📊 TFT_eSPI vs LVGL Comparison

### TFT_eSPI (What we used before)

```
Pros:
✓ Fast and lightweight
✓ Direct pixel control
✓ Smaller memory footprint
✓ Good for simple displays
✓ Easy to get started

Cons:
✗ Manual UI element management
✗ No built-in widgets
✗ More code for complex GUIs
✗ Harder to maintain large interfaces
✗ Manual touch handling
```

### LVGL (Professional choice for Bruce)

```
Pros:
✓✓ Professional-grade GUI framework
✓✓ Rich widget library (buttons, sliders, charts, etc.)
✓✓ Built-in animations and themes
✓✓ Object-oriented design
✓✓ Automatic touch handling
✓✓ Responsive layouts
✓✓ Visual UI designer (SquareLine Studio)
✓✓ Used in production embedded systems
✓✓ Active development & community

Cons:
✗ Larger memory usage (~200KB Flash, ~50KB RAM)
✗ Slightly slower than raw TFT_eSPI
✗ More complex initial setup
```

**Verdict for Bruce:** LVGL wins for professional intimidation factor! 🦈

## 🚀 LVGL Setup for Waveshare ESP32-S3 + DollaTek ILI9341

### Step 1: Install LVGL Library

**Arduino IDE:**

1. Open **Tools → Manage Libraries**
2. Search for "LVGL" by kisvegabor
3. Install **LVGL version 9.2** (latest)
4. Also install:
   - TFT_eSPI (for display driver - configure for ILI9341!)
   - XPT2046_Touchscreen (for touch)

**PlatformIO (recommended for larger projects):**

```ini
[env:esp32dev]
platform = espressif32
board = esp32dev
framework = arduino
lib_deps =
    bodmer/TFT_eSPI@^2.5.43
    paulstoffregen/XPT2046_Touchscreen@^1.4
    lvgl/lvgl@^9.2.0
build_flags =
    -DCORE_DEBUG_LEVEL=3
    -DBOARD_HAS_PSRAM
    -mfix-esp32-psram-cache-issue
monitor_speed = 115200
```

## 🦈 Bruce LVGL Main Code

### Complete Professional GUI Example

```cpp
/**
 * BRUCE - Professional LVGL GUI
 * ESP32-2432S028R Touchscreen
 */

#include <lvgl.h>
#include <TFT_eSPI.h>
#include <XPT2046_Touchscreen.h>

// Display & Touch
TFT_eSPI tft = TFT_eSPI();
#define TOUCH_CS 33
#define TOUCH_IRQ 36
XPT2046_Touchscreen touch(TOUCH_CS, TOUCH_IRQ);

// LVGL display buffer
#define SCREEN_WIDTH 240
#define SCREEN_HEIGHT 320
#define LVGL_BUFFER_SIZE (SCREEN_WIDTH * SCREEN_HEIGHT / 10)

static lv_disp_draw_buf_t draw_buf;
static lv_color_t buf1[LVGL_BUFFER_SIZE];
static lv_color_t buf2[LVGL_BUFFER_SIZE];

// Touch calibration
#define TOUCH_MIN_X 200
#define TOUCH_MAX_X 3700
#define TOUCH_MIN_Y 240
#define TOUCH_MAX_Y 3800

// RGB LED
#define LED_RED 4
#define LED_GREEN 16
#define LED_BLUE 17

// Bruce status
int battery_percent = 87;
int signals_found = 0;
const char* status_text = "ARMED";

// LVGL objects
lv_obj_t* screen_main;
lv_obj_t* label_status;
lv_obj_t* label_battery;
lv_obj_t* label_signals;
lv_obj_t* btn_hunt;
lv_obj_t* btn_scan;
lv_obj_t* btn_replay;
lv_obj_t* chart_spectrum;

// Bruce styles
static lv_style_t style_screen;
static lv_style_t style_btn_aggressive;
static lv_style_t style_btn_pressed;
static lv_style_t style_status_panel;

// Forward declarations
void lvgl_display_flush(lv_disp_drv_t* disp, const lv_area_t* area,
                        lv_color_t* color_p);
void lvgl_touchpad_read(lv_indev_drv_t* indev_driver, lv_indev_data_t* data);
void create_bruce_styles();
void create_bruce_gui();
void event_btn_hunt(lv_event_t* e);
void event_btn_scan(lv_event_t* e);
void event_btn_replay(lv_event_t* e);
void update_status_display();

void setup() {
    Serial.begin(115200);
    Serial.println("🦈 BRUCE: System initializing...");

    // Initialize LEDs
    pinMode(LED_RED, OUTPUT);
    pinMode(LED_GREEN, OUTPUT);
    pinMode(LED_BLUE, OUTPUT);
    setLED(0, 0, 255);  // Blue = ready

    // Initialize display
    tft.begin();
    tft.invertDisplay(true);
    tft.setRotation(0);  // Portrait
    tft.fillScreen(TFT_BLACK);

    // Initialize touch
    touch.begin();
    touch.setRotation(0);

    // Initialize LVGL
    lv_init();

    // Setup display driver
    static lv_disp_drv_t disp_drv;
    lv_disp_drv_init(&disp_drv);
    disp_drv.hor_res = SCREEN_WIDTH;
    disp_drv.ver_res = SCREEN_HEIGHT;
    disp_drv.flush_cb = lvgl_display_flush;
    disp_drv.draw_buf = &draw_buf;
    lv_disp_draw_buf_init(&draw_buf, buf1, buf2, LVGL_BUFFER_SIZE);
    lv_disp_drv_register(&disp_drv);

    // Setup touch driver
    static lv_indev_drv_t indev_drv;
    lv_indev_drv_init(&indev_drv);
    indev_drv.type = LV_INDEV_TYPE_POINTER;
    indev_drv.read_cb = lvgl_touchpad_read;
    lv_indev_drv_register(&indev_drv);

    // Create Bruce GUI
    create_bruce_styles();
    create_bruce_gui();

    Serial.println("🦈 BRUCE: Big brother ready. Let's dominate.");
}

void loop() {
    lv_timer_handler();  // LVGL task handler
    delay(5);

    // Update status periodically
    static unsigned long lastUpdate = 0;
    if (millis() - lastUpdate > 1000) {
        update_status_display();
        lastUpdate = millis();
    }
}

// Display flush callback
void lvgl_display_flush(lv_disp_drv_t* disp, const lv_area_t* area,
                        lv_color_t* color_p) {
    uint32_t w = (area->x2 - area->x1 + 1);
    uint32_t h = (area->y2 - area->y1 + 1);

    tft.startWrite();
    tft.setAddrWindow(area->x1, area->y1, w, h);
    tft.pushColors((uint16_t*)&color_p->full, w * h, true);
    tft.endWrite();

    lv_disp_flush_ready(disp);
}

// Touch read callback
void lvgl_touchpad_read(lv_indev_drv_t* indev_driver, lv_indev_data_t* data) {
    if (touch.touched()) {
        TS_Point p = touch.getPoint();

        // Map to screen coordinates
        data->point.x = map(p.x, TOUCH_MIN_X, TOUCH_MAX_X, 0, SCREEN_WIDTH);
        data->point.y = map(p.y, TOUCH_MIN_Y, TOUCH_MAX_Y, 0, SCREEN_HEIGHT);
        data->state = LV_INDEV_STATE_PRESSED;
    } else {
        data->state = LV_INDEV_STATE_RELEASED;
    }
}

// Create Bruce's main GUI
void create_bruce_gui() {
    screen_main = lv_obj_create(NULL);
    lv_obj_add_style(screen_main, &style_screen, 0);
    lv_scr_load(screen_main);

    // Header with shark emoji
    lv_obj_t* header = lv_obj_create(screen_main);
    lv_obj_set_size(header, 240, 40);
    lv_obj_set_pos(header, 0, 0);
    lv_obj_set_style_bg_color(header, lv_color_hex(0x1A1A1A), 0);

    lv_obj_t* label_title = lv_label_create(header);
    lv_label_set_text(label_title, "BRUCE v1.0");
    lv_obj_set_style_text_color(label_title, lv_color_hex(0xFFFFFF), 0);
    lv_obj_set_style_text_font(label_title, &lv_font_montserrat_20, 0);
    lv_obj_align(label_title, LV_ALIGN_LEFT_MID, 10, 0);

    lv_obj_t* label_shark = lv_label_create(header);
    lv_label_set_text(label_shark, "🦈");
    lv_obj_set_style_text_font(label_shark, &lv_font_montserrat_24, 0);
    lv_obj_align(label_shark, LV_ALIGN_RIGHT_MID, -10, 0);

    // Status panel
    lv_obj_t* status_panel = lv_obj_create(screen_main);
    lv_obj_add_style(status_panel, &style_status_panel, 0);
    lv_obj_set_size(status_panel, 220, 60);
    lv_obj_set_pos(status_panel, 10, 50);

    label_status = lv_label_create(status_panel);
    lv_label_set_text_fmt(label_status, "Status: %s", status_text);
    lv_obj_set_style_text_color(label_status, lv_color_hex(0x00FF00), 0);
    lv_obj_align(label_status, LV_ALIGN_TOP_LEFT, 5, 5);

    label_battery = lv_label_create(status_panel);
    lv_label_set_text_fmt(label_battery, "Battery: %d%%", battery_percent);
    lv_obj_set_style_text_color(label_battery, lv_color_hex(0x00FF00), 0);
    lv_obj_align(label_battery, LV_ALIGN_TOP_LEFT, 5, 25);

    // RF Spectrum chart
    chart_spectrum = lv_chart_create(screen_main);
    lv_obj_set_size(chart_spectrum, 220, 100);
    lv_obj_set_pos(chart_spectrum, 10, 120);
    lv_chart_set_type(chart_spectrum, LV_CHART_TYPE_LINE);
    lv_chart_set_range(chart_spectrum, LV_CHART_AXIS_PRIMARY_Y, 0, 100);
    lv_chart_set_point_count(chart_spectrum, 20);
    lv_obj_set_style_bg_color(chart_spectrum, lv_color_hex(0x000000), 0);

    // Add spectrum data
    lv_chart_series_t* ser = lv_chart_add_series(chart_spectrum,
                                                  lv_color_hex(0x00FFFF),
                                                  LV_CHART_AXIS_PRIMARY_Y);
    for (int i = 0; i < 20; i++) {
        lv_chart_set_next_value(chart_spectrum, ser, random(20, 80));
    }

    // Signals found
    label_signals = lv_label_create(screen_main);
    lv_label_set_text_fmt(label_signals, "Signals Found: %d", signals_found);
    lv_obj_set_style_text_color(label_signals, lv_color_hex(0xFFFFFF), 0);
    lv_obj_set_style_text_font(label_signals, &lv_font_montserrat_16, 0);
    lv_obj_set_pos(label_signals, 10, 230);

    // HUNT button
    btn_hunt = lv_btn_create(screen_main);
    lv_obj_add_style(btn_hunt, &style_btn_aggressive, 0);
    lv_obj_set_size(btn_hunt, 65, 50);
    lv_obj_set_pos(btn_hunt, 10, 260);
    lv_obj_add_event_cb(btn_hunt, event_btn_hunt, LV_EVENT_CLICKED, NULL);

    lv_obj_t* label_hunt = lv_label_create(btn_hunt);
    lv_label_set_text(label_hunt, "HUNT");
    lv_obj_set_style_text_color(label_hunt, lv_color_hex(0x000000), 0);
    lv_obj_set_style_text_font(label_hunt, &lv_font_montserrat_18, 0);
    lv_obj_center(label_hunt);

    // SCAN button
    btn_scan = lv_btn_create(screen_main);
    lv_obj_add_style(btn_scan, &style_btn_aggressive, 0);
    lv_obj_set_size(btn_scan, 65, 50);
    lv_obj_set_pos(btn_scan, 87, 260);
    lv_obj_add_event_cb(btn_scan, event_btn_scan, LV_EVENT_CLICKED, NULL);

    lv_obj_t* label_scan = lv_label_create(btn_scan);
    lv_label_set_text(label_scan, "SCAN");
    lv_obj_set_style_text_color(label_scan, lv_color_hex(0x000000), 0);
    lv_obj_set_style_text_font(label_scan, &lv_font_montserrat_18, 0);
    lv_obj_center(label_scan);

    // REPLAY button
    btn_replay = lv_btn_create(screen_main);
    lv_obj_add_style(btn_replay, &style_btn_aggressive, 0);
    lv_obj_set_size(btn_replay, 65, 50);
    lv_obj_set_pos(btn_replay, 165, 260);
    lv_obj_add_event_cb(btn_replay, event_btn_replay, LV_EVENT_CLICKED, NULL);

    lv_label_set_text(label_replay, "PLAY");

// Button event handlers
void event_btn_hunt(lv_event_t* e) {
    Serial.println("🦈 BRUCE: Hunting mode activated");
    status_text = "HUNTING";
    setLED(255, 255, 0);  // Yellow
    lv_label_set_text_fmt(label_status, "Status: %s", status_text);
}

void event_btn_scan(lv_event_t* e) {
    Serial.println("🦈 BRUCE: Sweeping all bands...");
    status_text = "SCANNING";
    setLED(255, 0, 0);  // Red
    signals_found++;
    lv_label_set_text_fmt(label_status, "Status: %s", status_text);
    lv_label_set_text_fmt(label_signals, "Signals Found: %d", signals_found);

void event_btn_replay(lv_event_t* e) {
    Serial.println("🦈 BRUCE: Replaying signal. Too easy.");
    status_text = "REPLAYING";
    setLED(0, 255, 0);  // Green
    lv_label_set_text_fmt(label_status, "Status: %s", status_text);
}

void update_status_display() {
    // Update battery (simulate drain)
    battery_percent = max(20, battery_percent - random(0, 2));
    lv_label_set_text_fmt(label_battery, "Battery: %d%%", battery_percent);

    // Update spectrum chart (simulate RF activity)
    lv_chart_series_t* ser = lv_chart_get_series_next(chart_spectrum, NULL);
    if (ser) {
        lv_chart_set_next_value(chart_spectrum, ser, random(20, 80));
    }
}

void setLED(int r, int g, int b) {
    analogWrite(LED_RED, 255 - r);
    analogWrite(LED_GREEN, 255 - g);
    analogWrite(LED_BLUE, 255 - b);
}
```
